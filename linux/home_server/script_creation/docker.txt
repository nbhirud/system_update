# Installs latest Docker Engine from official Docker repo (not old Ubuntu package).
# Adds current user (e.g. nik) to docker group.
# Installs latest docker-compose as a plugin.
# Enables Docker to start on boot.

# usage:
# chmod +x install-docker.sh
# sudo ./install-docker.sh nik   # Replace 'nik' with your actual username



#!/bin/bash

set -e

USER_TO_ADD=${1:-$SUDO_USER}

echo ">>> Updating system..."
sudo apt update && sudo apt install -y ca-certificates curl gnupg lsb-release

echo ">>> Setting up Docker GPG key and repo..."
sudo install -m 0755 -d /etc/apt/keyrings
curl -fsSL https://download.docker.com/linux/ubuntu/gpg | \
  sudo gpg --dearmor -o /etc/apt/keyrings/docker.gpg
sudo chmod a+r /etc/apt/keyrings/docker.gpg

echo \
  "deb [arch=$(dpkg --print-architecture) \
  signed-by=/etc/apt/keyrings/docker.gpg] \
  https://download.docker.com/linux/ubuntu \
  $(lsb_release -cs) stable" | \
  sudo tee /etc/apt/sources.list.d/docker.list > /dev/null

echo ">>> Installing Docker Engine..."
sudo apt update
sudo apt install -y docker-ce docker-ce-cli containerd.io docker-buildx-plugin docker-compose-plugin

echo ">>> Enabling Docker service..."
sudo systemctl enable docker
sudo systemctl start docker

echo ">>> Adding user '$USER_TO_ADD' to docker group..."
sudo usermod -aG docker $USER_TO_ADD

echo ">>> Verifying installation..."
docker version
docker compose version

echo ">>> Docker and Compose installed successfully. Reboot or log out/in to apply group membership."
